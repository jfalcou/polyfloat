using wf3_t = eve::wide<f3_t, 4>;

std::cout << "wxd " << wxd << std::endl;
std::cout << "sin(wxf) = " << sin(wxf) << std::endl;
std::cout << "sin(wxd) = " << sin(wxd) << std::endl;
wf2_t wxf2([](auto i, auto) { return f1_t(i / 20.0f); });
wd2_t wxd2([](auto i, auto) { return d1_t(i / 20.0f); });

std::cout << "sin(wf3_t(xf)) = " << sin(wf3_t(wxf)) << std::endl;
using pl2_t = plf::polyfloat<double, 2>;

std::cout << "plf::sin[eve::medium](x) = " << plf::sin[eve::medium](x) << std::endl;
std::cout << "eve::sin[eve::medium](-x) = " << eve::sin[eve::medium](-hx) << std::endl;
std::cout << "plf::sin[eve::medium](-x) = " << plf::sin[eve::medium](-x) << std::endl;

hx = plf::hi(x);

x += plf::pio_3(eve::as<pl2_t>());
hx = plf::hi(x);

std::cout << "eve::sin[eve::medium](x) = " << eve::sin[eve::medium](hx) << std::endl;
